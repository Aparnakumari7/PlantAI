                          BASIC QUERIES

-- View all tables
.tables

.schema users
.schema predictions

-- View all data
SELECT * FROM users;
SELECT * FROM predictions;

-- Count records
SELECT COUNT(*) FROM users;
SELECT COUNT(*) FROM predictions;


================================================================================
                          USER MANAGEMENT
================================================================================

-- View all users
SELECT id, username, email, is_admin, created_at FROM users;

-- Find specific user
SELECT * FROM users WHERE username = 'admin';
SELECT * FROM users WHERE email = 'user@example.com';

-- Make user admin
UPDATE users SET is_admin = 1 WHERE username = 'admin';

-- Remove admin privileges
UPDATE users SET is_admin = 0 WHERE id = 2;

-- Delete user
DELETE FROM users WHERE id = 1;

-- Count total users
SELECT COUNT(*) as total_users FROM users;

-- View admin users only
SELECT * FROM users WHERE is_admin = 1;


================================================================================
                          PREDICTION/SCAN QUERIES
================================================================================

-- View all predictions with user info
SELECT p.*, u.username, u.email 
FROM predictions p 
JOIN users u ON p.user_id = u.id 
ORDER BY p.timestamp DESC;

-- View recent 10 scans
SELECT * FROM predictions 
ORDER BY timestamp DESC 
LIMIT 10;

-- View specific user's scans
SELECT * FROM predictions WHERE user_id = 1;

-- Count healthy vs diseased
SELECT 
  SUM(CASE WHEN is_healthy = 1 THEN 1 ELSE 0 END) as healthy_count,
  SUM(CASE WHEN is_healthy = 0 THEN 1 ELSE 0 END) as diseased_count
FROM predictions;

-- Find specific disease
SELECT * FROM predictions 
WHERE disease_name LIKE '%blight%';

-- Most common diseases
SELECT disease_name, COUNT(*) as count
FROM predictions
WHERE is_healthy = 0 AND disease_name IS NOT NULL
GROUP BY disease_name
ORDER BY count DESC
LIMIT 10;

-- Most scanned plants
SELECT plant_name, COUNT(*) as count
FROM predictions
GROUP BY plant_name
ORDER BY count DESC
LIMIT 10;


================================================================================
                          STATISTICS QUERIES
================================================================================

-- Platform overview
SELECT 
  (SELECT COUNT(*) FROM users) as total_users,
  (SELECT COUNT(*) FROM predictions) as total_scans,
  (SELECT COUNT(*) FROM predictions WHERE is_healthy = 1) as healthy_scans,
  (SELECT COUNT(*) FROM predictions WHERE is_healthy = 0) as diseased_scans;

-- Scans per user
SELECT 
  u.username, 
  u.email,
  COUNT(p.id) as scan_count,
  SUM(CASE WHEN p.is_healthy = 1 THEN 1 ELSE 0 END) as healthy,
  SUM(CASE WHEN p.is_healthy = 0 THEN 1 ELSE 0 END) as diseased
FROM users u
LEFT JOIN predictions p ON u.id = p.user_id
GROUP BY u.id
ORDER BY scan_count DESC;

-- Daily scan activity
SELECT 
  DATE(timestamp) as date,
  COUNT(*) as scans
FROM predictions
GROUP BY DATE(timestamp)
ORDER BY date DESC
LIMIT 30;

-- Average confidence score
SELECT AVG(plant_confidence) as avg_confidence
FROM predictions;


================================================================================
                          DELETE OPERATIONS
================================================================================

-- Delete specific scan
DELETE FROM predictions WHERE id = 5;

-- Delete all scans for a user
DELETE FROM predictions WHERE user_id = 2;

-- Delete old scans (older than 30 days)
DELETE FROM predictions 
WHERE timestamp < datetime('now', '-30 days');

-- Delete all healthy scans
DELETE FROM predictions WHERE is_healthy = 1;

-- Clear all predictions (keep users)
DELETE FROM predictions;

-- Delete user and their scans
DELETE FROM predictions WHERE user_id = 3;
DELETE FROM users WHERE id = 3;

================================================================================
                          ADVANCED QUERIES
================================================================================

-- Users with most diseased plants
SELECT 
  u.username,
  COUNT(p.id) as diseased_count
FROM users u
JOIN predictions p ON u.id = p.user_id
WHERE p.is_healthy = 0
GROUP BY u.id
ORDER BY diseased_count DESC
LIMIT 10;

-- Disease severity distribution
SELECT 
  severity,
  COUNT(*) as count
FROM predictions
WHERE is_healthy = 0
GROUP BY severity;

-- Monthly scan trends
SELECT 
  strftime('%Y-%m', timestamp) as month,
  COUNT(*) as scans
FROM predictions
GROUP BY month
ORDER BY month DESC;

-- User registration trends
SELECT 
  DATE(created_at) as date,
  COUNT(*) as new_users
FROM users
GROUP BY DATE(created_at)
ORDER BY date DESC;

                          NOTES
================================================================================

1. Always backup database before making bulk changes
2. Use transactions for multiple operations
3. Be careful with DELETE operations - they cannot be undone
4. Use WHERE clause to avoid deleting all records
5. Test queries with SELECT before running UPDATE/DELETE
6. Keep database file backed up regularly
